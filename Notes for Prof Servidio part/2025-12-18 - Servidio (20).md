We assume quasi neutrality, stationarity and translational symmetry
$$
\begin{align}
 n_{i} \simeq n_{e} \equiv n; &&  \frac{ \partial  }{ \partial t } = 0; &&  \frac{ \partial }{ \partial y } = \frac{ \partial }{ \partial z } = 0
\end{align}

$$
f_{i,e} =  f_{i,e} (x,v_{x},v_{y},v_{z})

The Vlasov-Maxwell system is
$$
\begin{align}
 &\frac{ \partial f_{\alpha} }{ \partial t } + \bar{v} \cdot \nabla f_{\alpha} + \frac{q_{\alpha}}{m_{\alpha}} \left( \bar{E} + \frac{\bar{V}\times \bar{B}}{c} \right) \cdot \nabla_{\bar{v}} f_{\alpha} = 0 \\[.8em]
&\frac{ \partial ^2 A_{y}}{ \partial x^2 } = - \frac{4\pi}{c} e \left[  \int \bar{v} f_{i} \, d^3v - \int \bar{v} f_{e} \, d^3v \right] \cdot \hat{y} \\[.8em]
& \frac{ \partial ^2\phi }{ \partial x^2 } = -4\pi e \left[  \int f_{i} \, d^3v - \int f_{e} \, d^3v \right] 
\end{align}
$$
We take an integral of the motion: the Energy
$$
 \frac{d}{dt}  \mathscr{E} = 0
$$
if $f$ is a function of an invariant of motion, then it is a solution of the Vlasov problem
$$
\begin{align}
 f_{\alpha} = f_{\alpha} (\mathscr{E}) &&\implies&& \frac{D}{Dt} f_{\alpha} (\mathscr{E})= 0
\end{align}
$$
$f$ can be any function ($f_{\alpha} (\mathscr{E}) = \mathscr{E}$, $f_{\alpha} (\mathscr{E}) = \mathscr{E}^{\gamma}$ or $f_{\alpha} (\mathscr{E}) = \exp (\mathscr{E})$).
Proof (bases on the derivation chain rule):
$$
\begin{align}
 \frac{D}{Dt} f_{\alpha}(\mathscr{E}) & = \frac{ \partial f_{\alpha} (\mathscr{E}) }{ \partial t }  + \bar{v} \cdot \nabla f_{ \alpha} (\mathscr{E}) + \bar{a} \cdot \nabla_{\bar{v}} f_{\alpha} (\mathscr{E}) =  \\[.8em]
& = \frac{ \partial f_{\alpha} }{ \partial \mathscr{E} } \frac{ \partial \mathscr{E} }{ \partial t }  + \frac{ \partial f_{\alpha} }{ \partial \mathscr{E} } (\bar{v} \cdot \nabla \mathscr{E}) + \frac{ \partial f_{\alpha} }{ \partial \mathscr{E} } (\bar{a} \cdot \nabla_{\bar{v}}\, \mathscr{E}) =  \\[.8em]
& = \frac{ \partial f_{\alpha} }{ \partial \mathscr{E} } \left[ {\frac{ \partial \mathscr{E} }{ \partial t }  + \bar{v} \cdot \nabla \mathscr{E} + \bar{a} \cdot \nabla_{\bar{v}}\, \mathscr{E}} \right] =  \\[.8em]
&= \frac{ \partial f_{\alpha} }{ \partial \mathscr{E} } \left( \frac{D}{Dt} \mathscr{E} \right) = 0
\end{align}
$$
