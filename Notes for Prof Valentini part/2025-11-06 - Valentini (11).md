Recap 
$$
\omega_{R} = \frac{kc_{s}}{\sqrt{  1 + k^2 \lambda_{De}^2}} \left[ 1 + 3 \frac{T_{i 0}}{T_{e 0}} \right]^{1/2} 
$$
with 
$$
\begin{align}
\left| \frac{\omega_{I}}{\omega_{R}} \right| \ll 1, && \frac{T_{i 0}}{T_{ e 0}} \ll 1
\end{align}
$$
$$
D_{R}(k,\omega_{R}) = 1- \frac{\omega_{p i}^2}{\omega_{R}^2} \left[ 1+ \frac{3k^2\lambda_{Di}^2\omega_{p i}^2}{\omega_{R}^2} \right] + \frac{1}{k^2\lambda_{De}^2}
$$
$$
\omega_{I} = - \frac{D_{I}(k, \omega_{R})}{\partial D_{R} / \partial \omega_{R}}
$$
$$
\begin{align}
D_{I} (k, \omega_{R}) &= -\pi \left. \sum_{\alpha} \frac{\omega_{p \alpha}^2}{k^2 n_{\alpha 0}} \frac{ \partial F_{\alpha 0} (u) }{ \partial u } \right|_{\omega_{R} / k}  =\\[.8em]
&= -\pi\frac{\omega_{p i}^2}{k^2n_{0}} \left.\frac{ \partial F_{i 0} (u) }{ \partial u } \right|_{\omega_{R} / k} -\pi\frac{\omega_{p e}^2}{k^2n_{0}} \left.\frac{ \partial F_{e 0} (u) }{ \partial u } \right|_{\omega_{R} / k}
\end{align}
$$
In the low frequency regime we also have
$$
v_{th \, i} \ll \frac{\omega_{R}}{k} \ll v_{th \, e}
$$
We use the approximation
$$
\begin{align}
D_{R} (k, \omega_{R}) \approx 1 - \frac{\omega_{p i}^2}{\omega_{R}^2} + \frac{1}{k^2\lambda_{De}^2} && \implies &&  \frac{ \partial D_{R} }{ \partial \omega_{R} } \approx \frac{2\omega_{p i}^2}{\omega_{R}^3}
\end{align}
$$
Now we want to find $\omega_{I}$ 
$$
\begin{align}
\omega_{I} &= \frac{\pi}{k^2n_{0}} \frac{\omega_{R}^3}{2 \omega_{p i}^2} \left[ \omega_{p i}^2 \left.\frac{ \partial F_{i 0} (u) }{ \partial u } \right|_{\omega_{R} / k} + \omega_{p e}^2 \left.\frac{ \partial F_{e 0} (u) }{ \partial u } \right|_{\omega_{R} / k} \right] = \\[.8em]
&= \frac{\pi}{k^2n_{0}} \frac{\omega_{R}^3}{2} \left[ \left.\frac{ \partial F_{i 0} (u) }{ \partial u } \right|_{\omega_{R} / k} + \frac{\omega_{p e}^2}{\omega_{p i}^2} \left.\frac{ \partial F_{e 0} (u) }{ \partial u } \right|_{\omega_{R} / k} \right] 
\end{align}
$$
Now we use the maxwellian distribution and its derivative
$$
\begin{align}
F_{\alpha 0} (u) &= \frac{n_{0}m_{\alpha}^{ 1/2 }}{\left( 2\pi k_{B}T_{\alpha 0} \right) ^{1/2}} \exp \left( - \frac{m_{\alpha}}{2k_{B}T_{\alpha 0}} u^2 \right) \\[.8em]
\frac{ \partial F_{\alpha 0} (u)}{ \partial u } &=  \frac{n_{0}m_{\alpha}^{ 1/2 }}{\left( 2\pi k_{B}T_{\alpha 0} \right) ^{1/2}} \left( - \frac{m_{\alpha}u}{k_{B}T_{\alpha 0}}  \right) \exp \left( - \frac{m_{\alpha}}{2k_{B}T_{e 0}} u^2 \right) \\[.8em]
\left. \frac{ \partial F_{\alpha 0} (u)}{ \partial u } \right|_{\omega_{R} /k}&=  - \frac{n_{0}m_{\alpha}^{ 3/2 }}{\left( 2\pi  \right)^{1/2} \left(   k_{B}T_{\alpha 0} \right) ^{3/2}} \frac{\omega_{R}}{k} \exp \left( - \frac{m_{\alpha}}{2k_{B}T_{\alpha 0}} \frac{\omega_{R}^2}{k^2} \right)
\end{align}
$$
and
$$
\begin{align}
\omega_{I} &= \frac{\pi}{k^2n_{0}} \frac{\omega_{R}^3}{2} \left[ - \frac{n_{0}m_{i}^{ 3/2 }}{\left( 2\pi  \right)^{1/2} \left(   k_{B}T_{i 0} \right) ^{3/2}} \frac{\omega_{R}}{k} \exp \left( - \frac{m_{i}}{2k_{B}T_{i 0}} \frac{\omega_{R}^2}{k^2} \right) \right. + \\[.8em] 
&\ \ \ \ \ \left. - \frac{\omega_{p e}^2}{\omega_{p i}^2} \frac{n_{0}m_{e}^{ 3/2 }}{\left( 2\pi  \right)^{1/2} \left(   k_{B}T_{e 0} \right) ^{3/2}} \frac{\omega_{R}}{k} \exp \left( - \frac{m_{e}}{2k_{B}T_{e 0}} \frac{\omega_{R}^2}{k^2} \right) \right] = \\[.8em]
&= - \sqrt{ \frac{\pi}{8} } \frac{\omega_{R}^4m_{i}^{3/2}}{k^3 (k_{B}T_{e  0})^{3/2}} \left[ \left( \frac{T_{e 0}}{T_{i 0}} \right)^{3/2} \exp \left( - \frac{m_{i}}{2k_{B}T_{i 0}} \frac{\omega_{R}^2}{k^2} \right) + \right. \\[.8em]
&\ \ \ \ \ \left. + \frac{m_{i}}{m_{e}} \left( \frac{m_{e}}{m_{i}} \right)^{3/2}  \exp \left( - \frac{m_{e}}{2k_{B}T_{e 0}} \frac{\omega_{R}^2}{k^2} \right)  \right]  = \\[.8em]
&= - \sqrt{ \frac{\pi}{8} } \frac{\omega_{R}^4m_{i}^{3/2}}{k^3 (k_{B}T_{e  0})^{3/2}} \left[ \left( \frac{T_{e 0}}{T_{i 0}} \right)^{3/2} \exp \left( - \frac{m_{i}}{2k_{B}T_{i 0}} \frac{\omega_{R}^2}{k^2} \right) + \right. \\[.8em]
&\ \ \ \ \ \left. + \left( \frac{m_{e}}{m_{i}} \right)^{1/2} \exp \left( - \frac{m_{e}}{2k_{B}T_{e 0}} \frac{\omega_{R}^2}{k^2} \right)  \right] 
\end{align}
$$
Now, using
$$
\begin{align}
\frac{T_{i 0}}{T_{ e 0}} \ll 1  && \text{and} && \frac{m_{e}}{m_{i}} \ll 1
\end{align}
$$
we approximate 
$$
\exp \left( - \frac{m_{e}}{2k_{B}T_{e 0}} \frac{\omega_{R}^2}{k^2} \right) \approx 1
$$
and
$$
\begin{align}
\omega_{I} &= - \sqrt{ \frac{\pi}{8} } \frac{\omega_{R}^4}{k^3 c_{s}^3} \left[ \left( \frac{T_{e 0}}{T_{i 0}} \right)^{3/2} \exp \left( - \frac{m_{i}}{2k_{B}T_{i 0}} \frac{\omega_{R}^2}{k^2} \right) + \sqrt{ \frac{m_{e}}{m_{i}} }\,  \right] = \\[.8em]
&= - \sqrt{ \frac{\pi}{8} } \frac{\omega_{R}}{\cancel{ k^3 c_{s}^3 }} \frac{\cancel{ k^3c_{s}^3 }}{(1+ k^2 \lambda_{De}^2)^{3/2}} \left[ \left( \frac{T_{e 0}}{T_{i 0}} \right)^{3/2} \exp \left( - \frac{m_{i}}{2k_{B}T_{i 0}} \frac{\omega_{R}^2}{k^2} \cdot \frac{T_{e 0}}{T_{e 0}}\right) + \sqrt{ \frac{m_{e}}{m_{i}} }\,  \right] =  \\[.8em]
&= - \sqrt{ \frac{\pi}{8} } \frac{\omega_{R}}{(1+ k^2 \lambda_{De}^2)^{3/2}} \left[ \left( \frac{T_{e 0}}{T_{i 0}} \right)^{3/2} \exp \left( - \frac{T_{e 0}}{2T_{i 0}} \frac{1}{1+k^2\lambda_{De}^2} \right) + \sqrt{ \frac{m_{e}}{m_{i}} }\,  \right]
\end{align}
$$
where in the second passage we used
$$
\omega_{R} = \frac{ k^3c_{s}^3 }{(1+ k^2 \lambda_{De}^2)^{3/2}} \left[ 1 + 3 \frac{T_{i 0}}{T_{ e 0}} \right]^{1/2} \approx \frac{ k^3c_{s}^3 }{(1+ k^2 \lambda_{De}^2)^{3/2}} 
$$
In the last expression of $\omega_{I}$ we have the factor 
$$
\sqrt{ \frac{m_{e}}{m_{i}} } \sim \frac{1}{40}
$$
which is negligile.
Furthermore we can affirm that 
$$
\begin{align}
\frac{T_{i 0}}{T_{e 0}}\ll 1 && \implies &&  \left| \frac{\omega_{I}}{\omega_{R}} \right| \ll 1
\end{align}
$$
which in conclusion represent the same condition. 


## Bump-on-tail instability
$$
\begin{align}
D(k,\omega) &= 1- \sum_{\alpha} \frac{\omega_{p \alpha}^2}{k^2n_{\alpha 0}} \int_{L} \frac{\partial F_{\alpha 0}(u) / \partial u}{u - \omega / k}\, du \\[.8em]
 D_{R}(k, \omega_{R}) &= 1- \sum_{\alpha} \frac{\omega_{p \alpha}^2}{k^2n_{\alpha 0}} P \int_{-\infty}^{\infty} \frac{\partial F_{\alpha 0}(u) / \partial u}{u - \omega_{R} / k}\, du \\[.8em]
 D_{I}(k, \omega_{R}) &= - \pi  \sum_{\alpha} \frac{\omega_{p \alpha}^2}{k^2n_{\alpha 0}} \left. \frac{ \partial F_{\alpha 0}(u) }{ \partial u }  \right|_{\omega_{R} / k}
\end{align}
$$
High frequency regime
Propagation speed is too high for ions to follow the wave, so only electrons give a contribute.
PIC BumpOnTail
![[Screenshot from 2025-11-18 13-25-52.png|650]]
These symmetric bumps avoid the generation a current $\bar{J}$, hence also the magnetic field will be null (this is required by the electrostatic approximation in the derivation of Vlasov-Poisson equations).
$$
\begin{align}
|u_{0}| \gg v_{th\,1} = \sqrt{  \frac{k_{B}T_{1}}{m_{e}} }, &&\frac{\omega_{R}}{k} \gg v_{th\, 1} \gg v_{th\, 2}, &&T_{1} \gg T_{2}
\end{align}
$$
"gentle bump"
We also assume
$$
\begin{align}
n_{1} + 2 n_{2} \approx n_{1}, && \frac{n_{2}}{n_{1}} \ll 1
\end{align}
$$
and
$$
\begin{align}
k^2 \lambda_{De}^2 \ll 1, && \omega_{I} \propto \left. \frac{ \partial F_{e 0}(u) }{ \partial u } \right|_{\omega_{R} / k}
\end{align}
$$
We use the following distribution function
$$
\begin{align}
F_{e 0}(u) &= \frac{m_{e}^{1/2}n_{1}}{\left( 2\pi k_{B}T_{1} \right) ^{1/2}} \exp \left( - \frac{m_{e}}{2k_{B}T_{1} } u^2 \right)+  \\[.8em]
&\ \ \ \ \ + \frac{m_{e}^{1/2}n_{2}}{\left( 2\pi k_{B}T_{2} \right) ^{1/2}} \exp \left( - \frac{m_{e}}{2k_{B}T_{2} } (u-|u_{0}|)^2 \right)+  \\[.8em]
&\ \ \ \ \ + \frac{m_{e}^{1/2}n_{2}}{\left( 2\pi k_{B}T_{2} \right) ^{1/2}} \exp \left( - \frac{m_{e}}{2k_{B}T_{2} } (u+|u_{0}|)^2 \right)
\end{align}
$$
In the integral of $D_{R}(k, \omega_{R}) = 0$ we can neglect the derivative because it is local, the high value of the derivatives is due to the bumps, but it is negligible with respect to the rest of the integral. 
$$
\begin{align}
D_{R}(k, \omega_{R}) = 0 &&\implies&& \omega_{R} = \omega_{p 1} \sqrt{ 1+3k^2\lambda_{D 1}^2 } 
\end{align}
$$
and
$$
D_{R}(k, \omega_{R}) = 1-\frac{\omega_{p 1}^2}{\omega_{R}^2} - \frac{3k^2\lambda_{D 1}^2\omega_{p 1}^2}{\omega_{R}^2} = 0
$$
when we find the derivative of such quantity, we will neglect the last term ($k^2\lambda_{D 1}^2 \ll 1$) and
$$
\frac{ \partial D_{R} }{ \partial \omega_{R} } \approx \frac{2\omega_{p 1}^2}{\omega_{R}^3} 
$$
For the imaginary part, we cannot neglect the local derivative at the bumps. 
First we evaluate 
$$
\begin{align}
 \frac{ \partial F_{e 0}(u) }{ \partial u } &= \frac{m_{e}^{3/2}n_{1}}{\left( 2\pi \right)^{1/2} \left(   k_{B}T_{1} \right) ^{3/2}} u \exp \left( - \frac{m_{e}}{2k_{B}T_{1} } u^2 \right)+  \\[.8em]
&\ \ \ \ \ - \frac{m_{e}^{3/2}n_{2}}{\left( 2\pi \right)^{1/2} \left(   k_{B}T_{2} \right) ^{3/2}} \left\{ (u-|u_{0}|) \exp \left( - \frac{m_{e}}{2k_{B}T_{2} } (u-|u_{0}|)^2 \right)+ \right.  \\[.8em]
&\ \ \ \ \ \left. + (u+|u_{0}|) \exp \left( - \frac{m_{e}}{2k_{B}T_{2} } (u+|u_{0}|)^2 \right) \right\}
\end{align}
$$
Now we evaluate this quantity in $u =\omega_{R} / k$. 
We notice that the third exponential is similar to $\exp(-M)$ with $M$ a really big number, so this term tends to zero and we neglect it. 
We have
$$
\begin{align}
\left. \frac{ \partial F_{e 0}(u) }{ \partial u } \right|_{\omega_{R} / k} &= \frac{m_{e}^{3/2}n_{1}}{\left( 2\pi \right)^{1/2} \left(   k_{B}T_{1} \right) ^{3/2}} \frac{\omega_{R}}{k} \exp \left[ - \frac{m_{e}}{2k_{B}T_{1}} \left( \frac{\omega_{R}}{k} \right)^2 \right]+  \\[.8em]
&\ \ \ \ \ - \frac{m_{e}^{3/2}n_{2}}{\left( 2\pi \right)^{1/2} \left(   k_{B}T_{2} \right) ^{3/2}} \left( \frac{\omega_{R}}{k}-|u_{0}| \right) \exp \left[ - \frac{m_{e}}{2k_{B}T_{2} } \left( \frac{\omega_{R}}{k}-|u_{0}| \right)^2 \right]
\end{align}
$$
So 
$$
 \begin{align}
\omega_{I} &= -\frac{D_{I}}{ \partial D_{R}   / \partial \omega _{R}  } =\\[.8em]
 &= (-1) (-1) \frac{\omega_{R}^3}{2\omega_{p 1}^2} \pi \left\{ - \frac{\omega_{p 1}^2}{k^2\cancel{ n_{1} }} \frac{m_{e}^{3/2}\cancel{ n_{1} }\omega_{R}}{k \left( 2\pi \right)^{1/2} \left( k_{B}T_{1} \right)^{3/2}} \exp \left[ - \frac{m_{e}}{2k_{B}T_{1}} \left( \frac{\omega_{R}}{k} \right)^2 \right]  + \right.  \\[.8em]
&\ \ \ \ \ -\left. \frac{m_{e}^{3/2}\cancel{ n_{2} }\omega_{p 2}^2}{k^2\cancel{ n_{2} } \left( 2\pi \right)^{1/2} \left( k_{B}T_{2} \right)^{3/2}} \left( \frac{\omega_{R}}{k}-|u_{0}| \right) \exp \left[ - \frac{m_{e}}{2k_{B}T_{1}} \left( \frac{\omega_{R}}{k}-|u_{0}| \right)^2 \right] \right\} =  \\[.8em]
&=  \omega_{R}^3 \sqrt{ \frac{\pi}{8} } \frac{\cancel{ \omega_{p 1}^2 }}{\cancel{ \omega_{p 1}^2 }} \frac{m_{e}^{3/2}}{k^2 (k_{B}T_{1})^{3/2}} \left\{ - \frac{\omega_{R}}{k} \exp \left[ - \frac{m_{e}}{2k_{B}T_{1}} \left( \frac{\omega_{R}}{k} \right)^2 \right]  + \right.  \\[.8em]
&\ \ \ \ \ +\left. \frac{n_{2}}{n_{1}} \left( \frac{T_{1}}{T_{2}} \right)^{3/2} \left( |u_{0}|-\frac{\omega_{R}}{k} \right) \exp \left[ - \frac{m_{e}}{2k_{B}T_{2}} \left( |u_{0}| -\frac{\omega_{R}}{k} \right)^2 \right] \right\}
\end{align}
$$
Where we used the fact that 
$$
\frac{\omega_{R}}{k} < |u_{0}|
$$
Now we collect $\omega_{R} /k$ and
$$
 \begin{align}
\omega_{I} &=  \sqrt{ \frac{\pi}{8} }  \frac{\omega_{R}^4 m_{e}^{3/2}}{k^3 (k_{B}T_{1})^{3/2}} \left\{ - \exp \left[ - \frac{m_{e}}{2k_{B}T_{1}} \left( \frac{\omega_{R}}{k} \right)^2 \right]  + \right.  \\[.8em]
&\ \ \ \ \ +\left. \frac{n_{2}}{n_{1}} \left( \frac{T_{1}}{T_{2}} \right)^{3/2} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right) \exp \left[ - \frac{m_{e}}{2k_{B}T_{2}} \frac{\omega_{R}^2}{k^2} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right)^2 \right] \right\} = \\[.8em]
&=  \sqrt{ \frac{\pi}{8} }  \frac{\omega_{R} \omega_{p 1}^3}{k^3 v_{th \, 1}^3} \left\{ - \exp \left[ - \frac{m_{e}}{2k_{B}T_{1}}  \frac{\omega_{p 1}^2(1+3k^2\lambda_{D1}^2)}{2} \right]  + \right.  \\[.8em]
&\ \ \ \ \ +\left. \frac{n_{2}}{n_{1}} \left( \frac{T_{1}}{T_{2}} \right)^{3/2} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right) \exp \left[ - \frac{m_{e}}{2k_{B}T_{2}} \frac{T_{1}}{T_{1}} \frac{\omega_{p 1}^2}{k^2} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right)^2 \right] \right\} = \\[.8em]
&=  \sqrt{ \frac{\pi}{8} }  \frac{\omega_{R}}{k^3 \lambda_{D 1}^3} \left\{ - \exp \left[ - \frac{1}{2k^2 \lambda_{D 1}^2} - \frac{3}{2} \right]  + \right.  \\[.8em]
&\ \ \ \ \ +\left. \frac{n_{2}}{n_{1}} \left( \frac{T_{1}}{T_{2}} \right)^{3/2} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right) \exp \left[ - \frac{1}{2k^2 \lambda_{D 1}^2} \frac{T_{1}}{T_{2}} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right)^2 \right] \right\}
\end{align}
$$
We isolate 
$$
\omega_{I,\text{beam}} = \underbrace{ \sqrt{ \frac{\pi}{8} }  \frac{\omega_{R}}{k^3 \lambda_{D 1}^3} \frac{n_{2}}{n_{1}} \left( \frac{T_{1}}{T_{2}} \right)^{3/2} }_{ \beta } \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right) \exp \left[ - \frac{1}{2k^2 \lambda_{D 1}^2} \frac{T_{1}}{T_2} \left( \frac{|u_{0}|k}{\omega_{R}} -1 \right)^2 \right]
$$
We use the change of variable 
$$
y = \frac{|u_{0}|k}{\omega_{R}}
$$
and we want to find the maximum of this quantity
$$
\begin{align}
\frac{ \partial \omega_{I,\text{beam}} }{ \partial y } &= \frac{ \partial }{ \partial y } \left\{ \beta (y-1) \exp\left[ - \frac{1}{2k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} (y-1)^2 \right] \right\} = \\[.8em]
&= \beta \exp\left[ - \frac{1}{2k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} (y-1)^2 \right] + \\[.8em]
&\ \ \ \ + \beta (y-1) \left( - \frac{1}{k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} (y-1) \right)  \exp\left[ - \frac{1}{2k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} (y-1)^2 \right] = \\[.8em]
&= \beta \left( 1- \frac{\left( y-1 \right)^2}{k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} \right)  \exp\left[ - \frac{1}{2k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} (y-1)^2 \right] = 0
\end{align}
$$
We know that $\exp(x) \neq 0, \ \ \forall x \in \mathbb{R}$ so
$$
\begin{align}
&1- \frac{\left( y-1 \right)^2}{k^2\lambda_{D 1}^2} \frac{T_{1}}{T_{2}} = 0 &\implies& \left( y-1 \right)^2= \frac{T_{2}}{T_{1}} k^2 \lambda_{D 1}^2  \\[.8em]
 \implies& y-1 = \pm \sqrt{ \frac{T_{2}}{T_{1}} } k \lambda_{D 1}  &\implies& y = 1\pm \sqrt{ \frac{T_{2}}{T_{1}} } k \lambda_{D 1} \\[.8em]
\implies& \frac{|u_{0}|k}{\omega_{R}} = 1\pm \sqrt{ \frac{T_{2}}{T_{1}} } k \lambda_{D 1} &\implies& |u_{0}| = \frac{\omega_{R}}{k} \left( 1\pm\sqrt{ \frac{T_{2}}{T_{1}} } k \lambda_{D 1} \right)
\end{align}
$$
Between these two solution of $|u_{0}|$ we choose the one which satisfies the instability condition
($\omega_{I} > 0$) and which could lead to instabilities.

We substitute this definition into $\omega_{I}$ and 
$$
 \begin{align}
\omega_{I} &=  \sqrt{ \frac{\pi}{8} }  \frac{\omega_{R}}{k^3 \lambda_{D 1}^3} \left\{ - \exp \left[ - \frac{1}{2k^2 \lambda_{D 1}^2} - \frac{3}{2} \right]  + \right.  \\[.8em]
&\ \ \ \ \ +\left. \frac{n_{2}}{n_{1}} \left( \frac{T_{1}}{T_{2}} \right)^{3/2} \left( 1 -\sqrt{ \frac{T_{2}}{T_{1}} }k \lambda_{D1} -1 \right) \exp \left[ - \frac{1}{2k^2 \lambda_{D 1}^2} \frac{T_{1}}{T_{2}} \frac{T_{2}}{T_{1}} k^2 \lambda_{D 1}^2 \right] \right\} =  \\[.8em]
&= \sqrt{ \frac{\pi}{8} } \frac{\omega_{R}}{k^3\lambda_{D 1}^3} \left\{ \frac{n_{2}}{n_{1}} \frac{T_{1}}{T_{2}} k \lambda_{D 1} e^{ -1/2 } - \exp\left[ -\frac{1}{2k^2\lambda_{D 1}^2} - \frac{3}{2} \right] \right\} >0
\end{align}
$$
Which is the instability condition. 
To have an instability it is sufficient to have the maximum point as an instable point.

It's possible to repeat this for the Ion Acoustic Waves, with the following hypothesis
$$
\begin{align}
\left| \frac{\omega_{I}}{\omega_{R}} \right| \ll 1, && \frac{\omega_{R}}{k} \gg v_{th\, i}, && \frac{T_{e}}{T_{i}} \gg 1
\end{align}
$$
and
$$
\begin{align}
\left| \frac{\omega_{R}}{k} - u_{0} \right|   \ll v_{th\,e}, && (u_{0}>0)
\end{align}
$$
